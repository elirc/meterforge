sealed class RecordingProvider(bool failAfterRecording) : IPaymentProvider
{
    public List<ProviderChargeRequest> Requests { get; } = [];
    public Task<ProviderChargeResult> ChargeAsync(ProviderChargeRequest request, CancellationToken token)
    {
        token.ThrowIfCancellationRequested(); Requests.Add(request);
        if (failAfterRecording) throw new IOException("Synthetic effect recorded before acknowledgment loss");
        return Task.FromResult(new ProviderChargeResult(true, "synthetic-reference"));
    }
}
